\documentclass{ltjsarticle}
\usepackage{amsmath}
\usepackage{amsthm}
\usepackage{amssymb}
\usepackage{enumitem}

% 定理環境の定義
\newtheorem{thm}{Theorem}[subsection]
\newtheorem{prop}[thm]{Prop}

% 系用のカスタムスタイル
\newtheoremstyle{corollarystyle} % 新しいスタイルを定義
  {\topsep}    % 上のスペース
  {\topsep}    % 下のスペース
  {\itshape}   % 本文のフォント
  {}           % 見出しのインデント
  {\bfseries}  % 見出しのフォント
  {.}          % 見出しの後の句読点
  {.5em}       % 見出しと本文の間のスペース
  {\thmname{命題} \thmnumber{\arabic{section}.\arabic{subsection}.\arabic{thm}} 系}

\theoremstyle{corollarystyle}
\newtheorem{cor}[thm]{Prop} % 系を定義（命題と同じカウンターを共有）

\begin{document}

\section*{Example Section}
\refstepcounter{section} % セクション番号を手動で進める
\subsection*{Example Subsection}
\refstepcounter{subsection} % サブセクション番号を手動で進める

% Prop 1.3
\begin{prop}
This is Proposition 1.3.
\end{prop}
\begin{prop}
  This is Proposition 1.3.
\end{prop}
\begin{prop}
  This is Proposition 1.3.
\end{prop}

\setcounter{subsection}{5}
% Prop 1.3 系
\begin{cor}
This is Corollary 1.3 based on Proposition 1.3.
\end{cor}

% Prop 1.4
\begin{prop}
This is Proposition 1.4.
\end{prop}

% Prop 1.4 系
\begin{cor}
This is Corollary 1.4 based on Proposition 1.4.
\end{cor}

\end{document}
